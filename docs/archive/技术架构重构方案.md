#### 3.1 上下文窗口管理
```python
class ContextWindowManager:
    """上下文窗口管理器"""
    
    def __init__(self, max_tokens: int = 128000):
        self.max_tokens = max_tokens
        self.current_tokens = 0
        self.messages = []
        self.compressor = ContextCompressor()
    
    def add_message(self, role: str, content: str):
        """添加消息到上下文"""
        tokens = count_tokens(content)
        
        # 如果超出限制，压缩历史对话
        if self.current_tokens + tokens > self.max_tokens * 0.8:
            self.compress_context()
        
        self.messages.append({"role": role, "content": content})
        self.current_tokens += tokens
    
    def compress_context(self):
        """压缩上下文"""
        # 保留最近的3轮对话
        recent_messages = self.messages[-6:]
        
        # 压缩早期对话
        old_messages = self.messages[:-6]
        summary = self.compressor.summarize(old_messages)
        
        self.messages = [
            {"role": "system", "content": f"历史对话摘要：{summary}"}
        ] + recent_messages
    
    def get_context(self) -> List[Dict]:
        """获取当前上下文"""
        return self.messages
```

```python
class ConversationMemory:
    """对话记忆系统"""
    
    def __init__(self):
        self.vector_db = VectorDatabase()  # FAISS或Chroma
        self.sql_db = SQLiteDatabase()
    
    def save_conversation(self, 
                         session_id: str,
                         user_message: str,
                         assistant_message: str,
                         metadata: Dict):
        """保存对话"""
        # 1. 存储到SQLite（结构化数据）
        self.sql_db.insert_conversation(
            session_id=session_id,
            user_msg=user_message,
            assistant_msg=assistant_message,
            timestamp=datetime.now(),
            metadata=metadata
        )
        
        # 2. 向量化并存储（语义检索）
        combined_text = f"用户: {user_message}\nAI: {assistant_message}"
        embedding = self.vector_db.embed(combined_text)
        self.vector_db.add(
            text=combined_text,
            embedding=embedding,
            metadata={"session_id": session_id, **metadata}
        )
    
    def retrieve_relevant_context(self, query: str, top_k: int = 3) -> List[Dict]:
        """检索相关上下文"""
        # 语义相似度搜索
        results = self.vector_db.search(query, top_k=top_k)
        
        return [
            {
                "text": r["text"],
                "similarity": r["score"],
                "metadata": r["metadata"]
            }
            for r in results
        ]
    
    def get_session_history(self, session_id: str, limit: int = 10) -> List[Dict]:
        """获取会话历史"""
        return self.sql_db.query_session(session_id, limit=limit)
```

#### 3.3 精确快速检索
```python
class HybridRetriever:
    """混合检索器：向量检索 + 关键词检索"""
    
    def __init__(self):
        self.vector_store = FAISSVectorStore()
        self.keyword_index = KeywordIndex()  # 基于倒排索引
    
    def retrieve(self, query: str, top_k: int = 5) -> List[Dict]:
        """混合检索"""
        # 1. 向量检索（语义相似）
        vector_results = self.vector_store.search(query, top_k=top_k)
        
        # 2. 关键词检索（精确匹配）
        keyword_results = self.keyword_index.search(query, top_k=top_k)
        
        # 3. 融合排序（RRF算法）
        merged_results = self.merge_results(vector_results, keyword_results)
        
        return merged_results[:top_k]
    
    def merge_results(self, vec_results: List, kw_results: List) -> List:
        """融合结果（Reciprocal Rank Fusion）"""
        scores = defaultdict(float)
        
        for rank, item in enumerate(vec_results, 1):
            scores[item["id"]] += 1 / (60 + rank)
        
        for rank, item in enumerate(kw_results, 1):
            scores[item["id"]] += 1 / (60 + rank)
        
        # 按分数排序
        sorted_ids = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return [{"id": id, "score": score} for id, score in sorted_ids]
```

#### 4.2 RAG Pipeline
```python
class RAGPipeline:
    """RAG检索增强生成管道"""
    
    def __init__(self):
        self.retriever = HybridRetriever()
        self.reranker = CrossEncoderReranker()  # 重排序
        self.generator = LLMGenerator()
    
    async def query(self, 
                   user_query: str, 
                   context_type: str = "all",
                   top_k: int = 5) -> str:
        """RAG查询流程"""
        
        # 1. 检索相关文档
        retrieved_docs = self.retriever.retrieve(
            query=user_query,
            context_type=context_type,
            top_k=top_k * 2  # 检索2倍，用于重排序
        )
        
        # 2. 重排序（提高相关性）
        reranked_docs = self.reranker.rerank(
            query=user_query,
            documents=retrieved_docs,
            top_k=top_k
        )
        
        # 3. 构建增强Prompt
        context = self.build_context(reranked_docs)
        
        prompt = f"""
基于以下知识库信息回答用户问题：

【知识库内容】
{context}

【用户问题】
{user_query}

【回答要求】
1. 基于知识库内容回答
2. 如果知识库没有相关信息，明确说明
3. 引用来源（如：根据《腾讯报价单模板》...）
"""
        
        # 4. 生成回答
        response = await self.generator.generate(prompt)
        
        # 5. 添加引用
        response_with_citations = self.add_citations(response, reranked_docs)
        
        return response_with_citations
    
    def build_context(self, docs: List[Dict]) -> str:
        """构建上下文"""
        context_parts = []
        for i, doc in enumerate(docs, 1):
            context_parts.append(f"""
[文档{i}] {doc['source']}
{doc['content']}
---
""")
        return "\n".join(context_parts)
```

#### 5.4 Tool + MCP + Skill 联动
```python
class AgentOrchestrator:
    """Agent编排器 - 协调Tool、MCP、Skill"""
    
    def __init__(self):
        self.tools = ToolRegistry()
        self.mcp_server = MCPServer()
        self.skills = SkillRegistry()
        self.llm = UniversalLLMClient()
    
    async def execute_task(self, task: str, context: Dict) -> Dict:
        """执行任务 - 智能选择Tool、MCP或Skill"""
        
        # 1. 让LLM分析任务
        analysis = await self.llm.analyze_task(task, context)
        
        # 2. 判断使用什么
        if analysis["task_type"] == "simple_tool_call":
            # 简单工具调用
            tool_name = analysis["recommended_tool"]
            result = await self.tools.execute(tool_name, **context)
        
        elif analysis["task_type"] == "external_service":
            # 外部服务调用（MCP）
            service_name = analysis["service_name"]
            connector = self.mcp_server.get_connector(service_name)
            result = await connector.call(analysis["method"], **context)
        
        elif analysis["task_type"] == "complex_workflow":
            # 复杂流程（Skill）
            skill_name = analysis["recommended_skill"]
            skill = self.skills.get(skill_name)
            result = await skill.execute(context)
        
        elif analysis["task_type"] == "multi_step":
            # 多步骤任务 - 组合使用
            result = await self.execute_multi_step(analysis["steps"], context)
        
        return result
    
    async def execute_multi_step(self, steps: List[Dict], context: Dict) -> Dict:
        """执行多步骤任务"""
        results = []
        
        for step in steps:
            if step["type"] == "tool":
                result = await self.tools.execute(step["name"], **context)
            elif step["type"] == "mcp":
                connector = self.mcp_server.get_connector(step["service"])
                result = await connector.call(step["method"], **context)
            elif step["type"] == "skill":
                skill = self.skills.get(step["name"])
                result = await skill.execute(context)
            
            results.append(result)
            # 将结果传递给下一步
            context.update(result)
        
        return {"steps_results": results, "final_context": context}
```

```python
class AgentHarness:

    def __init__(self):

        self.shared_memory = SharedMemory()
    
    def register_agent(self, agent_id: str, agent: BaseAgent):
        """注册Agent"""

    async def run_multi_agent(self, tasks: List[Dict]) -> Dict:
        """运行多个Agent协作"""
        results = await asyncio.gather(*[
            self.run_agent(task["agent_id"], task["task"])
            for task in tasks
        ])
        return {"results": results}
```

```python
        self.locks = {}
```

```python
class TokenBudgetManager:
    """Token预算管理"""
    
    def __init__(self):
        self.limits = {
            "daily": 100000,
            "monthly": 2000000,
            "per_request": 8000
        }
        self.alerts = []
    
    def check_limit(self, requested_tokens: int) -> bool:
        """检查是否超限"""
        current_usage = self.get_current_usage()
        
        if current_usage["daily"] + requested_tokens > self.limits["daily"]:
            self.alerts.append("日限额即将超限")
            return False
        
        return True
    
    def get_remaining_budget(self) -> Dict:
        """获取剩余预算"""
        current = self.get_current_usage()
        return {
            "daily_remaining": self.limits["daily"] - current["daily"],
            "monthly_remaining": self.limits["monthly"] - current["monthly"]
        }
```
